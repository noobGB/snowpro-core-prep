# Crawl directives for https://snowpro.gauravbarwalia.com
#
# Shipped via ROOT_ASSETS in app/vite.config.ts rather than Vite's normal public/ handling --
# publicDir is repurposed to ../content (the pipeline output), so this file needs the same
# hand-copy treatment the favicons get. See that file's own comment for the full reasoning.
#
# Note this one file also ships in the image self-hosters run, so a LAN box serves a Sitemap:
# line pointing at the public domain. Harmless (no crawler reaches a LAN box), and it's the same
# tradeoff app/index.html already documents for its hardcoded absolute og:image URL.
#
# IMPORTANT if you edit this: a crawler obeys exactly ONE group -- the most specific one matching
# its token -- and does NOT inherit rules from `User-agent: *`. So the Disallow set has to be
# repeated in every group, not stated once at the top. The named AI crawlers below are therefore
# listed as consecutive User-agent lines sharing a single rule block (valid per the spec) rather
# than as one group each, which would have silently granted them everything `*` is denied.

User-agent: *
Allow: /
# Never useful to a crawler, and /admin + /reset-password are auth/token surfaces.
Disallow: /api/
Disallow: /admin
Disallow: /reset-password
# The generated content bundle is the practice/mock question bank *including every answer key and
# explanation*. It is served unauthenticated on purpose -- the same questions live in public
# markdown in this MIT-licensed repo, and gating it while that is true would be theater. Keeping it
# out of search indexes and training corpora is the control that actually means something here.
Disallow: /content.json
Disallow: /notes/
Disallow: /search-index.json

# Answer-engine and AI crawlers, named explicitly. Several of these operators document that they
# look for their own token, and an explicit Allow is the signal they publish -- this site does want
# to be read and cited by answer engines. The rules match `*` exactly (see the note above on why
# they must be repeated rather than inherited).
User-agent: GPTBot
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: ClaudeBot
User-agent: Claude-User
User-agent: Claude-SearchBot
User-agent: anthropic-ai
User-agent: PerplexityBot
User-agent: Perplexity-User
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: Bingbot
User-agent: CCBot
Allow: /
Disallow: /api/
Disallow: /admin
Disallow: /reset-password
Disallow: /content.json
Disallow: /notes/
Disallow: /search-index.json
